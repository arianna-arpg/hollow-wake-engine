cmake_minimum_required(VERSION 3.25)
project(hwe_save LANGUAGES C CXX)
if(NOT DEFINED SQLITE_SOURCE OR NOT DEFINED HWE_OUTPUT)
  message(FATAL_ERROR "SQLITE_SOURCE and HWE_OUTPUT are required")
endif()
foreach(file IN ITEMS sqlite3.c sqlite3.h)
  if(NOT EXISTS "${SQLITE_SOURCE}/${file}")
    message(FATAL_ERROR "Missing SQLite amalgamation: ${file}")
  endif()
endforeach()
find_package(Threads REQUIRED)
add_library(hwe_save_sqlite OBJECT "${SQLITE_SOURCE}/sqlite3.c")
target_compile_definitions(hwe_save_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_THREADSAFE=1 SQLITE_DQS=0)
set_target_properties(hwe_save_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
add_library(hwe_save SHARED src/hwe_save.cpp src/slot-store.cpp $<TARGET_OBJECTS:hwe_save_sqlite>)
target_include_directories(hwe_save PUBLIC include PRIVATE "${SQLITE_SOURCE}")
target_link_libraries(hwe_save PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_compile_definitions(hwe_save PRIVATE HWE_SAVE_BUILD PUBLIC HWE_SAVE_SHARED)
set_target_properties(hwe_save PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${HWE_OUTPUT}" RUNTIME_OUTPUT_DIRECTORY "${HWE_OUTPUT}" ARCHIVE_OUTPUT_DIRECTORY "${HWE_OUTPUT}")
if(MSVC)
  set_property(TARGET hwe_save hwe_save_sqlite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(hwe_save PRIVATE /W4 /WX /utf-8)
  target_compile_options(hwe_save_sqlite PRIVATE /W0)
else()
  target_compile_options(hwe_save PRIVATE -Wall -Wextra -Werror)
endif()
