cmake_minimum_required(VERSION 3.25)
project(hwe_godot LANGUAGES CXX)
if(NOT DEFINED GODOT_CPP_SOURCE OR NOT DEFINED HWE_NATIVE_LIBRARY OR NOT DEFINED HWE_OUTPUT)
    message(FATAL_ERROR "Pinned Godot source, verified native library and output directory are required")
endif()
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${GODOT_CPP_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp" SYSTEM)
add_library(hwe_native SHARED IMPORTED)
set_target_properties(hwe_native PROPERTIES IMPORTED_LOCATION "${HWE_NATIVE_LIBRARY}")
if(WIN32)
    set_target_properties(hwe_native PROPERTIES IMPORTED_IMPLIB "${HWE_NATIVE_IMPORT_LIBRARY}")
endif()
add_library(hwe_godot SHARED host.cpp)
target_link_libraries(hwe_godot PRIVATE godot-cpp hwe_native)
target_include_directories(hwe_godot PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../embedded-js/native")
set_target_properties(hwe_godot PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    LIBRARY_OUTPUT_DIRECTORY "${HWE_OUTPUT}" RUNTIME_OUTPUT_DIRECTORY "${HWE_OUTPUT}"
    BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(MSVC)
    target_compile_options(hwe_godot PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(hwe_godot PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
