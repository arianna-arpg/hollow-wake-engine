private readonly sdkSkillDispatch = new SkillDispatch<Actor, SkillInstance, Vec2, number,
    ResolvedTarget, ReturnType<typeof skillContextTags>, ReturnType<typeof instanceMods>, Zone>({
    preflight: (actor, inst, aim, seatPress) => this.sdkSkillRequest.preflight(actor, inst, aim, seatPress),
    time: () => this.time,
    failNote: (actor, key, text) => this.failNote(actor, key, text),
    canAct: actor => actor.canAct(),
    canUse: (actor, inst) => actor.canUse(inst),
    comboCursor: (actor, inst) => this.comboCursorOf(actor, inst),
    definition: id => SKILLS[id],
    comboStep: (actor, inst, index) => this.comboStepOf(actor, inst, index),
    dispatch: (actor, inst, aim, seatPress) => this.useSkill(actor, inst, aim, seatPress),
    copyAim: aim => vec(aim.x, aim.y),
    facingTo: (actor, aim) => angleTo(actor.pos, aim),
    status: id => STATUS_DEFS[id],
    random: () => Math.random(),
    applyStatus: (actor, id, source, duration, label) => { actor.applyStatus(id, source, duration, label); },
    scrambleLatch: () => this.scrambleLatch,
    setScrambleLatch: value => { this.scrambleLatch = value; },
    trigger: instanceTrigger,
    feedback: (actor, kind, color) => {
      switch (kind) {
        case 'scramble': this.text(vec(actor.pos.x, actor.pos.y - 16), 'addled!', color, 12, 'combat'); break;
        case 'strobe-off': this.text(actor.pos, 'the earth rests', color, 11); break;
        case 'hex-off': this.text(actor.pos, 'hex sheathed', color, 11, 'combat'); break;
        case 'hex-on': this.text(actor.pos, 'hex drawn', color, 11, 'combat'); break;
      }
    },
    deactivateAura: (actor, id) => this.deactivateAura(actor, id),
    dismissSummon: (actor, id) => this.dismissSummonToggle(actor, id),
    maxMana: actor => actor.maxMana(),
    cost: (actor, inst) => actor.skillCost(inst),
    pay: (actor, inst, cost) => this.paySkillCost(actor, inst, cost),
    curseOnHit: inst => socketSpec(inst, 'curseOnHit'),
    curseField: instanceCurseField,
    zones: () => this.zones,
    expireZone: zone => this.expireZone(zone),
    castMode: instanceCastMode,
    tags: def => skillContextTags(def),
    modifiers: instanceMods,
    stat: (actor, key, tags, mods, base) => actor.sheet.get(key, tags, mods, base),
    useTime: (actor, inst) => actor.skillUseTime(inst),
    targeting: instanceTargeting,
    resolveTarget: (actor, inst, aim) => this.resolveTargeting(actor, inst, aim),
    lightweightCount: () => this.lite.liveCount,
    promoteTarget: (actor, aim, within, max) => this.litePromoteNearest(aim, {within, max, opposing: actor.team, story: actor.tier}),
    useCharges: instanceUseCharges,
    ammunition: () => this.sdkAmmunitionPress,
    channel: instanceChannel,
    gather: inst => socketSpec(inst, 'gather'),
    speed: (actor, inst) => actor.speedFactor(inst),
    consumeCharges: (actor, inst) => this.consumeChargeCost(actor, inst),
    spendGauge: (actor, inst) => gaugeSpend(inst, inst.def.gauge!, actor.gaugeEff(inst)!),
    stampCooldown: (actor, inst, duration) => this.stampSkillCooldown(actor, inst, duration),
    opening: (actor, inst, aim, multiplier) => this.attackSequences.opening(actor, inst, aim, multiplier),
    quarry: target => target.actor,
    guardCapacity,
    refreshGuard: (actor, state) => this.refreshGuardBash(actor, state),
    guardRaised: (actor, inst) => this.guardArts.raised(actor, inst),
    shellGraft: inst => socketSpec(inst, 'shellGraft'),
    overcharge: instanceOvercharge,
    strikeTiming: instanceStrikeTiming,
    assaultLock: (actor, inst, aim) => this.assaults.lock(actor, inst, aim),
    primedPourOpen: (actor, inst) => actor.primedPourOpen(inst),
    execute: (actor, inst, aim, payload) => this.executeSkill(actor, inst, aim, payload),
    isReflex: (actor, inst) => actor.isReflex(inst),
    reflexRecovery: () => REFLEX_CFG.lock,
  });
