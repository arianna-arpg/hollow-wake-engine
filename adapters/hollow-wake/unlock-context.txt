
import {UnlockLifecycle} from __RUNTIME__;
import {
 LEDGER_QUESTS_COMPLETED, LEDGER_QUEST_DONE_PREFIX, LEDGER_VOCATION_PREFIX,
 questDoneKey, vocationUnlockKey,
} from './account';
const engineUnlocks = new UnlockLifecycle<Account>({
 get catalog(){ return UNLOCK_CATALOG; },
 unlock: id => CATALOG_BY_ID.get(id),
 get slotTiers(){ return SLOT_TIERS; },
 get classBundles(){ return CLASS_BUNDLES; },
 get starterSkills(){ return STARTER_SKILLS; },
 get memoryConfig(){ return MEMORY_UNLOCK_CFG; },
 get memoryDefinitions(){ return MEMORY_UNLOCKS; },
 get classWeb(){ return CLASS_WEB_CFG; },
 gates: {
  reachedLevelKey: level => reachedLevelKey(level),
  classLevelLedgerKey: (id, level) => classLevelLedgerKey(id, level),
  vocationUnlockKey: id => vocationUnlockKey(id),
  questDoneKey: id => questDoneKey(id),
  get vocationPrefix(){ return LEDGER_VOCATION_PREFIX; },
  get questDonePrefix(){ return LEDGER_QUEST_DONE_PREFIX; },
  get questsCompletedKey(){ return LEDGER_QUESTS_COMPLETED; },
  className: id => CLASSES.find(row => row.id === id)?.name,
 },
 get reliquaryFeature(){ return FEATURE.RELIQUARY; },
 get reliquaryConfig(){ return RELIQUARY_CFG; },
 get relicStash(){ return RELIC_STASH; },
 get tabs(){ return VAULT_TABS; },
 get shelfConfig(){ return VAULT_SHELF_CFG; },
 get commissionNeed(){ return VENDOR_CFG.commission.need; },
 packages: () => PACKAGES,
 package: id => PACKAGE_BY_ID[id],
 classBundleId: id => classBundleId(id),
 resurrectUnlockId: id => resurrectUnlockId(id),
 isClassUnlocked: (a, id) => isClassUnlocked(a, id),
 unlockedClassCount: a => unlockedClassCount(a),
 encipher: text => encipher(text),
 revealScript: (text, fraction) => revealScript(text, fraction),
 awakeningRefusal: () => powerProgressionRefusal('awakening'),
 memoryCatalog: () => memoryCatalog(),
 memoryCommissionReady: (a, kind, id, need) => memoryCommissionReady(a, kind, id, need),
 memoryProgressionOpen: a => memoryProgressionOpen(a),
 memoryUnlockDef: id => memoryUnlockDef(id),
 memoryUnlockCandidates: (a, definition) => memoryUnlockCandidates(a, definition),
 grantMemoryUnlock: (a, definition, random) => grantMemoryUnlock(a, definition, random),
 reliquaryCost: r => reliquaryCost(r),
 reliquaryPower: r => reliquaryPower(r),
 stashPageCost: s => stashPageCost(RELIC_STASH, s),
 investReliquary: (a, amount) => investReliquary(a, amount),
 investRelicStash: (a, amount) => investRelicStash(a, amount),
});
