private readonly sdkSkillRequest = new SkillRequest<Actor, SkillInstance, Seat, Vec2>({
    castSealed: (actor, inst) => this.castSealed(actor, inst),
    hivecallMeta: (actor, inst) => this.hivecallMeta(actor, inst),
    throngTravelProtected,
    replenishingDelivery,
    forgetReplenishment: (actor, inst) => this.replenishment.forget(actor, inst),
    seatOf: actor => this.seatOf(actor),
    triggerSeat: actor => this.seats.find(seat => seat.actor === actor),
    markMetaDirty: seat => this.markMetaDirty(seat),
    markCharacterDirty: () => { this.charDirty = true; },
    replenishmentFeedback: (actor, inst) => this.text(vec(actor.pos.x, actor.pos.y - 20), inst.replenishmentPaused ? 'replenishment paused' : 'replenishment enabled', inst.def.color, 13),
    markSeatActed: actor => this.markSeatActed(actor),
    isReflex: (actor, inst) => actor.isReflex(inst),
    actorById: id => this.actorById(id),
    mashPress: () => GRAB_CFG.break.mashPress,
    failNote: (actor, key, text) => this.failNote(actor, key, text),
    maxLife: actor => actor.maxLife(),
    trigger: instanceTrigger,
    triggerArmed: instanceTriggerArmed,
    requirementRefusal: (actor, inst) => this.castReqRefusal(actor, inst),
    conversion: instanceConvert,
    definition: id => SKILLS[id],
    conversionHolds: (when, actor, inst) => convertRuleHolds(when, actor, inst, this),
    mint: (actor, host, id) => this.mintMetaInstance(actor, host, id),
    dispatch: (actor, inst, aim, seatPress) => this.useSkill(actor, inst, aim, seatPress),
    time: () => this.time,
    mimicSelected,
    mimicPowerMods: inst => mimicPowerMods(inst.def.mimic!),
    unmetGate: (actor, inst) => actor.unmetGate(inst),
    effects: instanceEffects,
    relocationTarget: (actor, inst, radius) => this.relocationTarget(actor, inst, radius),
    canUse: (actor, inst) => actor.canUse(inst),
  });
