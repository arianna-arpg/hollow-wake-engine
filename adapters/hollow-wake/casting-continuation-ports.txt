private readonly sdkCastingContinuation = new CastingContinuation<Actor, SkillInstance, CastingState,
    Vec2, ResolvedTarget, ReturnType<typeof skillContextTags>, ReturnType<typeof instanceMods>>({
    stunned: actor => actor.isStunned(),
    lowerGuard: (actor, inst) => this.guardArts.lower(actor, inst),
    feedback: (actor, state, kind) => {
      const cue = castingEventFlash(actor, state, kind);
      if (cue) this.flashes.push(cue);
    },
    updateAim: updateCastAim,
    seated: actor => !!this.seatOf(actor),
    primary: actor => actor === this.player,
    actor: id => this.actorById(id),
    assignAim: (aim, position) => { aim.x = position.x; aim.y = position.y; },
    copyAim: aim => vec(aim.x, aim.y),
    channel: instanceChannel,
    random: () => Math.random(),
    hold: state => this.aiHoldOf(state),
    needsFireLine: (actor, inst) => this.aiNeedsFireLine(actor, inst),
    lineOfFire: (actor, prey) => this.lineOfFire(actor.pos, prey.pos, actor.tier),
    finesse: actor => actor.brain ? normalizeBrain(actor.brain).base.skillUse?.finesse : undefined,
    click: actor => this.castPress(actor),
    distance: dist,
    tags: inst => skillContextTags(inst.def),
    mods: instanceMods,
    stat: (actor, id, tags, mods) => actor.sheet.get(id, tags, mods),
    target: (pos, actor) => actor ? {pos, actor} : {pos},
    deliver: (actor, inst, aim, options) => this.executeSkill(actor, inst, aim, options),
    advanceGuard: (actor, state, dt) => this.sdkGuardCasting().advance(actor, state, dt),
    advanceChannel: (actor, state, dt, target) => this.sdkChannelCasting().advance(actor, state, dt, target),
    advanceCharged: (actor, dt) => this.sdkChargedCasting().advance(actor, dt),
    plantTotem: (actor, inst, aim) => { this.plantSpiritTotem(actor, inst, aim); },
    spawnChannelVessel: (actor, inst, spec, aim) => this.spawnConstruct(actor, inst, spec, aim, inst),
    acted: actor => this.markSeatActed(actor),
    timingBonus: inst => instanceStrikeTiming(inst)?.bonus,
    timingCue: (actor, cue, aim) => { this.flashes.push(timingCueFlash(actor, cue, aim)); },
    flawless: (actor, aim, inst) => this.rollTriggers(actor, 'flawless', {aim, sourceInst: inst}),
  }, {channelGrace: LOS_CFG.channelGrace, focusSlack: CONCENTRATION_CFG.slack});
